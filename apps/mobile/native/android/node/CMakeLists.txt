cmake_minimum_required(VERSION 3.22)
project(anyframe_node)
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libnode.so")
add_library(anyframe_node SHARED bridge.cpp)
target_include_directories(anyframe_node PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include/node")
target_compile_features(anyframe_node PRIVATE cxx_std_20)
target_link_options(anyframe_node PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(anyframe_node node log android)
